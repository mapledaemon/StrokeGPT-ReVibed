# Optional isolated ASR runtime for Settings > Voice > NVIDIA Parakeet.
# Prefer scripts/install_parakeet.ps1 so NeMo dependencies stay out of .venv.
# Install CUDA-enabled PyTorch separately for practical latency on NVIDIA GPUs.
nemo_toolkit[asr]
# Some NeMo transitive imports still read NumPy 1.x compatibility attributes
# such as np.sctypes. Keep the isolated Parakeet runtime on the compatible
# NumPy line until that stack is fully NumPy 2 ready.
numpy>=1.26,<2
# NeMo 2.7.x pins fsspec exactly, while its Lightning integration needs a
# newer setuptools than some resolver paths leave behind. CUDA PyTorch 2.11
# rejects setuptools 82.0.1+, so cap this below 82 while staying new enough
# for NeMo's Lightning path.
fsspec==2024.12.0
setuptools>=79.0.0,<82
# ONNX 1.19+ ships Python protobuf modules generated with protobuf 6.31.x.
# Keep the runtime new enough so importing NeMo's ONNX path does not fail with
# "gencode 6.31.1 runtime 5.29.6" style version errors. The installer also
# force-reinstalls this after NeMo resolution because metadata checks can pass
# while ONNX's generated protobuf modules still fail at import time.
protobuf>=6.31.1,<7
